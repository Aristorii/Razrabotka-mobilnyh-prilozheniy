## Задание №2

**Код**
```phyton
fun main() {
    val a = 10
    val b = 4
    val c = a / b                    // 2 — целочисленное деление Int/Int
    val d = a / b.toDouble()         // 2.5 — один операнд Double, результат Double
    val e = "10" + 4                 // "104" — первый операнд строка, конкатенация
    val f = 10 + "4"                 // "104" — в Kotlin число + строка тоже конкатенация
    
    println("$c | $d | $e | $f")     // 2 | 2.5 | 104 | 104
}
```
**Вывод и объяснение**

2 | 2.5 | 104 | 104

Объяснение:

a / b равно 2, потому что оба числа целые (Int), и при делении дробная часть отбрасывается.

a / b.toDouble() равно 2.5, потому что b приводится к дробному типу (Double), и результат деления тоже становится дробным

"10" + 4 равно "104", потому что первый операнд — строка, поэтому число 4 превращается в строку и просто приклеивается в конец

10 + "4" равно "104", потому что в Kotlin, если хотя бы один операнд — строка, операция + всегда работает как склеивание строк (конкатенация)

```phyton
        println("Ошибка: город обязателен")
        return

    println("=== АНКЕТА ===")
```

**Код**

```phyton
fun line(left: String, right: String, width: Int = 32): String {
    val dots = width - left.length - right.length
    return left + ".".repeat(if (dots > 0) dots else 0) + right
}

fun task10() {
    print("Напиток: ")
    val rawName = readln().trim()
    if (rawName.isEmpty()) {
        println("Ошибка: название обязательно")
        return
    }

    print("Цена за чашку: ")
    val price = readln().replace(',', '.').trim().toDoubleOrNull()
    if (price == null || price <= 0) {
        println("Ошибка: некорректная цена")
        return
    }

    print("Количество чашек: ")
    val cups = readln().trim().toIntOrNull()
    if (cups == null || cups <= 0) {
        println("Ошибка: некорректное количество")
        return
    }

    print("Чаевые (%): ")
    val tipPercent = readln().trim().toIntOrNull()
    if (tipPercent == null || tipPercent !in 0..30) {
        println("Ошибка: чаевые должны быть от 0 до 30%")
        return
    }

    // Обрезка длинного названия: до 18 символов + ..
    val name = if (rawName.length > 20) rawName.take(18) + ".." else rawName

    val sum = price * cups
    val tip = sum * tipPercent / 100.0
    val total = sum + tip

    val sep = "=".repeat(32)
    println(sep)
    println(line("$name ", " ${"%.2f".format(sum)}"))
    println(line("Чашек: $cups x ", " ${"%.2f".format(price)}"))
    println(line("Сумма: ", " ${"%.2f".format(sum)}"))
    println(line("Чаевые ($tipPercent%): ", " ${"%.2f".format(tip)}"))
    println(sep)
    println(line("ИТОГО: ", " ${"%.2f".format(total)}"))
    println(sep)
}
```
